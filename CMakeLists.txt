cmake_minimum_required(VERSION 3.16)
project(fastdl C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "FastDL targets Linux HLDS")
endif()
option(FASTDL_BUILD_MODULE "Build the i386 Authum module" ON)
option(FASTDL_SANITIZE "Enable ASan/UBSan" OFF)
set(AUTHUM_DIR "${CMAKE_CURRENT_SOURCE_DIR}/lib/authum" CACHE PATH "Authum source tree")
find_package(Threads REQUIRED)
add_library(fastdl_http_core STATIC src/http/server.c src/fastdl/config.c lib/picohttpparser/picohttpparser.c)
target_include_directories(fastdl_http_core PUBLIC include "${AUTHUM_DIR}/include" PRIVATE lib/picohttpparser)
target_compile_definitions(fastdl_http_core PUBLIC _FILE_OFFSET_BITS=64)
target_compile_options(fastdl_http_core PRIVATE -Wall -Wextra -Werror -fvisibility=hidden -fstack-protector-strong -ffunction-sections -fdata-sections $<$<CONFIG:Release>:-Os>)
set_target_properties(fastdl_http_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(fastdl_http_core PUBLIC Threads::Threads)
if(FASTDL_BUILD_MODULE)
  target_compile_options(fastdl_http_core PUBLIC -m32)
  target_link_options(fastdl_http_core PUBLIC -m32)
  add_subdirectory("${AUTHUM_DIR}/client" "${CMAKE_BINARY_DIR}/authum-client")
  target_link_libraries(fastdl_http_core PUBLIC authum_client)
  add_library(fastdl MODULE src/authum/info.c src/authum/module.c src/authum/bridge.c)
  target_compile_options(fastdl PRIVATE -Wall -Wextra -Werror -fvisibility=hidden -ffunction-sections -fdata-sections $<$<CONFIG:Release>:-Os>)
  target_link_options(fastdl PRIVATE -static-libgcc "LINKER:--gc-sections" "LINKER:-z,relro,-z,now")
  target_link_libraries(fastdl PRIVATE authum_client fastdl_http_core)
  set_target_properties(fastdl PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/modules/fastdl")
  if(CMAKE_BUILD_TYPE MATCHES "^(Release|MinSizeRel)$" AND CMAKE_STRIP AND NOT FASTDL_SANITIZE)
    add_custom_command(TARGET fastdl POST_BUILD COMMAND ${CMAKE_STRIP} --strip-unneeded $<TARGET_FILE:fastdl>)
  endif()
  install(TARGETS fastdl LIBRARY DESTINATION addons/authum/modules/fastdl COMPONENT fastdl)
  install(FILES "${AUTHUM_DIR}/LICENSE" DESTINATION addons/authum/modules/fastdl/licenses RENAME authum-LICENSE COMPONENT fastdl)
  install(FILES "${AUTHUM_DIR}/lib/lh/LICENSE.md" DESTINATION addons/authum/modules/fastdl/licenses RENAME lh-LICENSE.md COMPONENT fastdl)
endif()
if(NOT FASTDL_BUILD_MODULE)
  # Only the test driver needs SDK wrappers in a host-free native build.
  add_library(fastdl_test_thread_client STATIC "${AUTHUM_DIR}/src/authum/api/thread.c")
  target_include_directories(fastdl_test_thread_client PUBLIC "${AUTHUM_DIR}/include")
  target_link_libraries(fastdl_http_core PUBLIC fastdl_test_thread_client)
endif()
if(FASTDL_SANITIZE)
  target_compile_options(fastdl_http_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(fastdl_http_core PUBLIC -fsanitize=address,undefined)
endif()
install(FILES fastdl.cfg DESTINATION addons/authum/modules/fastdl/configs COMPONENT fastdl)
install(FILES lib/picohttpparser/picohttpparser.h DESTINATION addons/authum/modules/fastdl/licenses COMPONENT fastdl)
install(FILES LICENSE DESTINATION addons/authum/modules/fastdl/licenses RENAME fastdl-LICENSE COMPONENT fastdl)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_library(fastdl_test_thread_host STATIC "${AUTHUM_DIR}/src/authum/host/threads.c")
  target_include_directories(fastdl_test_thread_host PUBLIC "${AUTHUM_DIR}/include" "${AUTHUM_DIR}/src")
  target_link_libraries(fastdl_test_thread_host PUBLIC Threads::Threads)
  if(FASTDL_BUILD_MODULE)
    target_compile_options(fastdl_test_thread_host PUBLIC -m32)
    target_link_options(fastdl_test_thread_host PUBLIC -m32)
  endif()
  if(FASTDL_SANITIZE)
    target_compile_options(fastdl_test_thread_host PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(fastdl_test_thread_host PUBLIC -fsanitize=address,undefined)
  endif()
  # Test driver only: production installs a single .so, no helper executable.
  add_executable(fastdl_test_http tests/http_driver.c)
  target_compile_options(fastdl_test_http PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(fastdl_test_http PRIVATE fastdl_http_core fastdl_test_thread_host)
  add_executable(fastdl_test_threads "${AUTHUM_DIR}/tests/threads.c")
  target_link_libraries(fastdl_test_threads PRIVATE fastdl_http_core fastdl_test_thread_host)
  add_test(NAME fastdl_threads COMMAND fastdl_test_threads)
  set_tests_properties(fastdl_threads PROPERTIES TIMEOUT 10)
  add_test(NAME fastdl_http COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/http.py $<TARGET_FILE:fastdl_test_http>)
  set_tests_properties(fastdl_http PROPERTIES TIMEOUT 45)
  if(FASTDL_BUILD_MODULE)
    add_executable(fastdl_test_module tests/module.c)
    target_compile_options(fastdl_test_module PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(fastdl_test_module PRIVATE authum_client fastdl_http_core fastdl_test_thread_host dl)
    add_test(NAME fastdl_module COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/module.py $<TARGET_FILE:fastdl_test_module> $<TARGET_FILE:fastdl>)
    set_tests_properties(fastdl_module PROPERTIES TIMEOUT 40)
  endif()
endif()
